"""Pure evaluation of WHERE expressions against a row.

Semantics:

* ``Literal`` -> its value; ``ColumnRef`` -> ``row[schema.index_of(name)]``
  (``UnknownColumnError`` for missing columns).
* ``Comparison``: NULL on either side -> NULL (None). Otherwise both sides
  must have the same ``sql_type_name`` or ``TypeMismatchError`` is raised.
  INTEGER and TEXT support all six operators; BOOLEAN supports all six
  with False < True.
* ``IsNull``: True/False, never NULL.
* ``Not`` / ``And`` / ``Or``: Kleene three-valued logic. Operands must be
  bool or None; anything else raises ``TypeMismatchError``.
"""

from __future__ import annotations

from minidb.types import (
    And,
    ColumnRef,
    Comparison,
    Expr,
    IsNull,
    Literal,
    Not,
    Or,
    Row,
    Schema,
    TypeMismatchError,
    Value,
    sql_type_name,
)

__all__ = [
    "evaluate",
    "compare",
    "logical_not",
    "logical_and",
    "logical_or",
    "where_matches",
]


def evaluate(expr: Expr, row: Row, schema: Schema) -> Value:
    """Evaluate ``expr`` against ``row`` laid out per ``schema``.

    Raises:
        UnknownColumnError: for a ColumnRef not in ``schema``.
        TypeMismatchError: for mixed-type comparisons or non-boolean
            logical operands.
    """
    raise NotImplementedError("evaluate not implemented")


def compare(op: str, left: Value, right: Value) -> Value:
    """Apply comparison ``op`` (one of ``COMPARISON_OPS``) to two values.

    Returns None if either side is None, else a bool.

    Raises:
        TypeMismatchError: if the operands' SQL types differ.
    """
    raise NotImplementedError("compare not implemented")


def logical_not(value: Value) -> Value:
    """Three-valued NOT: True -> False, False -> True, None -> None.

    Raises:
        TypeMismatchError: for a non-boolean, non-NULL operand.
    """
    raise NotImplementedError("logical_not not implemented")


def logical_and(left: Value, right: Value) -> Value:
    """Three-valued AND: False if either is False, else None if either is
    None, else True.

    Raises:
        TypeMismatchError: for a non-boolean, non-NULL operand.
    """
    raise NotImplementedError("logical_and not implemented")


def logical_or(left: Value, right: Value) -> Value:
    """Three-valued OR: True if either is True, else None if either is
    None, else False.

    Raises:
        TypeMismatchError: for a non-boolean, non-NULL operand.
    """
    raise NotImplementedError("logical_or not implemented")


def where_matches(expr: Expr, row: Row, schema: Schema) -> bool:
    """Return True only if ``expr`` evaluates to exactly True for ``row``;
    False for False or NULL.

    Raises:
        TypeMismatchError: if the result is a non-boolean, non-NULL value.
    """
    raise NotImplementedError("where_matches not implemented")

"""Unit tests for minidb.expr_eval."""

from __future__ import annotations

import pytest

from minidb.expr_eval import (
    compare,
    evaluate,
    logical_and,
    logical_not,
    logical_or,
    where_matches,
)
from minidb.types import (
    And,
    ColumnDef,
    ColumnRef,
    ColumnType,
    Comparison,
    IsNull,
    Literal,
    Not,
    Or,
    Schema,
    TypeMismatchError,
    UnknownColumnError,
)

SCHEMA = Schema(
    (
        ColumnDef("id", ColumnType.INTEGER),
        ColumnDef("name", ColumnType.TEXT),
        ColumnDef("active", ColumnType.BOOLEAN),
    )
)
ROW = (2, "bob", False)
NULL_ROW = (None, None, None)


def test_literal_and_column_ref():
    assert evaluate(Literal(5), ROW, SCHEMA) == 5
    assert evaluate(ColumnRef("name"), ROW, SCHEMA) == "bob"
    assert evaluate(ColumnRef("active"), ROW, SCHEMA) is False


def test_unknown_column_raises():
    with pytest.raises(UnknownColumnError) as excinfo:
        evaluate(ColumnRef("age"), ROW, SCHEMA)
    assert excinfo.value.column == "age"


@pytest.mark.parametrize(
    "op,left,right,expected",
    [
        ("=", 1, 1, True),
        ("=", 1, 2, False),
        ("<>", 1, 2, True),
        ("<", 1, 2, True),
        ("<=", 2, 2, True),
        (">", 3, 2, True),
        (">=", 1, 2, False),
        ("<", "alice", "bob", True),
        ("=", "a", "a", True),
        ("=", True, True, True),
        ("<>", True, False, True),
    ],
)
def test_compare_same_types(op, left, right, expected):
    assert compare(op, left, right) is expected


def test_compare_null_yields_null():
    assert compare("=", None, 1) is None
    assert compare("<", "a", None) is None
    assert compare("=", None, None) is None


def test_compare_mixed_types_raises():
    with pytest.raises(TypeMismatchError):
        compare("=", 1, "1")
    with pytest.raises(TypeMismatchError):
        compare("=", True, 1)
    with pytest.raises(TypeMismatchError):
        compare("<", 0, False)


def test_logical_not():
    assert logical_not(True) is False
    assert logical_not(False) is True
    assert logical_not(None) is None
    with pytest.raises(TypeMismatchError):
        logical_not(1)


def test_logical_and_kleene():
    assert logical_and(True, True) is True
    assert logical_and(True, False) is False
    assert logical_and(None, False) is False
    assert logical_and(False, None) is False
    assert logical_and(None, True) is None
    assert logical_and(None, None) is None
    with pytest.raises(TypeMismatchError):
        logical_and("x", True)


def test_logical_or_kleene():
    assert logical_or(False, False) is False
    assert logical_or(None, True) is True
    assert logical_or(True, None) is True
    assert logical_or(None, False) is None
    assert logical_or(None, None) is None
    with pytest.raises(TypeMismatchError):
        logical_or(False, 0)


def test_evaluate_comparison_on_row():
    expr = Comparison(">=", ColumnRef("id"), Literal(2))
    assert evaluate(expr, ROW, SCHEMA) is True
    assert evaluate(expr, NULL_ROW, SCHEMA) is None


def test_is_null_never_null():
    assert evaluate(IsNull(ColumnRef("name")), NULL_ROW, SCHEMA) is True
    assert evaluate(IsNull(ColumnRef("name"), negated=True), NULL_ROW, SCHEMA) is False
    assert evaluate(IsNull(ColumnRef("name")), ROW, SCHEMA) is False
    assert evaluate(IsNull(ColumnRef("name"), negated=True), ROW, SCHEMA) is True


def test_evaluate_nested_logic():
    expr = Or(Not(ColumnRef("active")), Comparison("=", ColumnRef("id"), Literal(1)))
    assert evaluate(expr, ROW, SCHEMA) is True
    assert evaluate(expr, (3, "c", True), SCHEMA) is False
    assert evaluate(expr, (3, "c", None), SCHEMA) is None


def test_not_on_non_boolean_column_raises():
    with pytest.raises(TypeMismatchError):
        evaluate(Not(ColumnRef("name")), ROW, SCHEMA)
    with pytest.raises(TypeMismatchError):
        evaluate(And(ColumnRef("id"), ColumnRef("active")), ROW, SCHEMA)


def test_where_matches_true_false_null():
    assert where_matches(Comparison("=", ColumnRef("id"), Literal(2)), ROW, SCHEMA) is True
    assert where_matches(Comparison("=", ColumnRef("id"), Literal(3)), ROW, SCHEMA) is False
    assert where_matches(Literal(None), ROW, SCHEMA) is False
    assert where_matches(Comparison("=", ColumnRef("id"), Literal(2)), NULL_ROW, SCHEMA) is False


def test_where_matches_bare_boolean_column():
    assert where_matches(ColumnRef("active"), (1, "a", True), SCHEMA) is True
    assert where_matches(ColumnRef("active"), ROW, SCHEMA) is False


def test_where_matches_non_boolean_raises():
    with pytest.raises(TypeMismatchError):
        where_matches(ColumnRef("id"), ROW, SCHEMA)
    with pytest.raises(TypeMismatchError):
        where_matches(Literal("x"), ROW, SCHEMA)


def test_compare_with_null_is_false():
    assert compare("=", None, 1) is False

"""MiniDB: an in-memory relational query engine for a small SQL subset.

Usage::

    from minidb import Engine

    engine = Engine()
    engine.execute("CREATE TABLE users (id INTEGER, name TEXT)")

``Engine`` is resolved lazily so that importing ``minidb.types`` (or any
other submodule) never drags in the full pipeline.
"""

from __future__ import annotations

from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from minidb.engine import Engine

__all__ = ["Engine"]


def __getattr__(name: str) -> object:
    if name == "Engine":
        from minidb.engine import Engine

        return Engine
    raise AttributeError(f"module 'minidb' has no attribute {name!r}")


=== SHARED SCHEMA: minidb/types.py ===
"""Shared types, AST nodes, constants, and exceptions for MiniDB.

Every MiniDB module imports from ``minidb.types`` using the full dotted
path. This module holds data definitions and tiny type helpers only; it
contains no tokenizing, parsing, evaluation, or storage logic.

Naming rules shared by every module:

* Table names and column names are case-sensitive and matched exactly.
* Keywords are case-insensitive; the tokenizer normalizes them to
  uppercase.

WHERE expression grammar (implemented by ``minidb.parser``, evaluated by
``minidb.expr_eval``)::

    expr       := or_expr
    or_expr    := and_expr ( OR and_expr )*          -> Or(left, right), left-assoc
    and_expr   := not_expr ( AND not_expr )*         -> And(left, right), left-assoc
    not_expr   := NOT not_expr | predicate           -> Not(operand)
    predicate  := primary [ comp_op primary          -> Comparison(op, left, right)
                          | IS [NOT] NULL ]          -> IsNull(operand, negated)
    primary    := INTEGER | STRING | TRUE | FALSE | NULL   -> Literal(value)
                | IDENTIFIER                              -> ColumnRef(name)
                | '(' expr ')'
    comp_op    := '=' | '<>' | '<' | '<=' | '>' | '>='

Comparisons do not chain: ``a = b = c`` is a syntax error. ``NOT a = 1``
parses as ``Not(Comparison('=', ColumnRef('a'), Literal(1)))``.

Statement grammar (keywords shown uppercase; trailing ';' optional)::

    CREATE TABLE name '(' col TYPE (',' col TYPE)* ')'
    DROP TABLE name
    INSERT INTO name [ '(' col (',' col)* ')' ] VALUES '(' literal (',' literal)* ')'
    SELECT ( '*' | col (',' col)* ) FROM name [ WHERE expr ]
    DELETE FROM name [ WHERE expr ]

where TYPE is one of INTEGER, TEXT, BOOLEAN and literal is an INTEGER,
STRING, TRUE, FALSE, or NULL token. Negative integer literals are not
supported.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from enum import Enum
from typing import Callable, Iterator, Optional, Protocol, Union

# ---------------------------------------------------------------------------
# Scalar values and rows
# ---------------------------------------------------------------------------

#: A scalar SQL value: INTEGER -> int, TEXT -> str, BOOLEAN -> bool,
#: NULL -> None. ``bool`` is a subclass of ``int`` in Python, but MiniDB
#: treats them as distinct: ``True`` is NOT a valid INTEGER and ``1`` is
#: NOT a valid BOOLEAN.
Value = Union[int, str, bool, None]

#: A stored or projected row: one Value per column, in column order.
Row = tuple[Value, ...]


# ---------------------------------------------------------------------------
# Exceptions
# ---------------------------------------------------------------------------


class MiniDBError(Exception):
    """Base class for every error MiniDB raises deliberately.

    ``str(error)`` is a single human-readable line; the CLI prints it as
    ``ERROR: <message>``.
    """


class SQLSyntaxError(MiniDBError):
    """Raised by the tokenizer for input it cannot lex, by the parser for
    token sequences that do not form a valid statement or expression, and
    by the engine for blank input or multiple statements passed to
    ``Engine.execute``."""


class UnknownTableError(MiniDBError):
    """Raised when a statement references a table that does not exist."""

    def __init__(self, table: str) -> None:
        super().__init__(f"unknown table: {table}")
        self.table = table


class UnknownColumnError(MiniDBError):
    """Raised when a column name does not exist in the relevant schema."""

    def __init__(self, column: str) -> None:
        super().__init__(f"unknown column: {column}")
        self.column = column


class TypeMismatchError(MiniDBError):
    """Raised when a value's type does not fit its context: an INSERT
    value that does not match the column type, an INSERT with the wrong
    number of values, a comparison between different SQL types, or a
    non-boolean operand to NOT/AND/OR/WHERE."""


class TableExistsError(MiniDBError):
    """Raised by CREATE TABLE when the table name is already in use."""

    def __init__(self, table: str) -> None:
        super().__init__(f"table already exists: {table}")
        self.table = table


# ---------------------------------------------------------------------------
# Schema
# ---------------------------------------------------------------------------


class ColumnType(Enum):
    """Declared column type. ``.value`` is the SQL keyword spelling."""

    INTEGER = "INTEGER"
    TEXT = "TEXT"
    BOOLEAN = "BOOLEAN"


@dataclass(frozen=True)
class ColumnDef:
    """One column of a table schema."""

    name: str
    type: ColumnType


@dataclass(frozen=True)
class Schema:
    """Ordered column definitions for a table (declaration order)."""

    columns: tuple[ColumnDef, ...]

    def column_names(self) -> list[str]:
        """Return column names in declaration order."""
        return [column.name for column in self.columns]

    def has_column(self, name: str) -> bool:
        """Return True if a column with exactly this name exists."""
        return any(column.name == name for column in self.columns)

    def index_of(self, name: str) -> int:
        """Return the zero-based position of column ``name``.

        Raises:
            UnknownColumnError: if no column has exactly this name.
        """
        for index, column in enumerate(self.columns):
            if column.name == name:
                return index
        raise UnknownColumnError(name)

    def __len__(self) -> int:
        return len(self.columns)


def value_matches_type(value: Value, column_type: ColumnType) -> bool:
    """Return True if ``value`` may be stored in a column of ``column_type``.

    None (NULL) matches every type. bool matches only BOOLEAN; int (but
    not bool) matches only INTEGER; str matches only TEXT.
    """
    if value is None:
        return True
    if column_type is ColumnType.BOOLEAN:
        return isinstance(value, bool)
    if column_type is ColumnType.INTEGER:
        return isinstance(value, int) and not isinstance(value, bool)
    if column_type is ColumnType.TEXT:
        return isinstance(value, str)
    return False


def sql_type_name(value: Value) -> str:
    """Return the SQL type name of a value: 'NULL', 'BOOLEAN', 'INTEGER',
    or 'TEXT'. Used for same-type checks and error messages."""
    if value is None:
        return "NULL"
    if isinstance(value, bool):
        return "BOOLEAN"
    if isinstance(value, int):
        return "INTEGER"
    return "TEXT"

class TokenType(Enum):
    """Kinds of lexical tokens."""

    KEYWORD = "KEYWORD"
    IDENTIFIER = "IDENTIFIER"
    INTEGER = "INTEGER"
    STRING = "STRING"
    OPERATOR = "OPERATOR"
    EOF = "EOF"


@dataclass(frozen=True)
class Token:
    """A lexical token.

    ``value`` by type:

    * KEYWORD: the keyword in uppercase, e.g. ``'SELECT'`` (always a member
      of ``KEYWORDS``).
    * IDENTIFIER: the name exactly as written, e.g. ``'users'``.
    * INTEGER: the digit text, e.g. ``'42'``.
    * STRING: the contents without the surrounding quotes, with ``''``
      unescaped to ``'``, e.g. ``'it''s'`` -> ``"it's"``.
    * OPERATOR: the symbol, one of ``OPERATORS``.
    * EOF: the empty string ``''``.

    ``position`` is the zero-based character offset of the token's first
    character in the source (the source length for EOF). It is excluded
    from equality so tests can compare tokens without positions.
    """

    type: TokenType
    value: str
    position: int = field(default=0, compare=False)

        "VALUES",
        "SELECT",
        "FROM",
        "WHERE",
        "DELETE",
        "AND",

        "IS",
        "NULL",
        "TRUE",

        "INTEGER",
