=== STUB FILE: minidb/tokenizer.py ===
"""Lexer for MiniDB's SQL subset.

``tokenize`` turns SQL text into a list of ``Token`` objects terminated by
exactly one EOF token. Lexical rules:

* Whitespace is skipped.
* A word starting with a letter or ``_`` and continuing with letters,
  digits, or ``_`` is a KEYWORD (emitted uppercase) if its uppercase form
  is in ``KEYWORDS``; otherwise an IDENTIFIER kept exactly as written.
* A run of ASCII digits is an INTEGER token whose value is the digit text.
* A single-quoted string is a STRING token; ``''`` inside it is unescaped
  to ``'``. A missing closing quote raises SQLSyntaxError.
* Operators are matched longest-first from ``OPERATORS``.
* Any other character raises SQLSyntaxError.
* Each token's ``position`` is the offset of its first character; the EOF
  token's position is ``len(sql)`` and its value is ``''``.
"""

from __future__ import annotations

from minidb.types import KEYWORDS, OPERATORS, SQLSyntaxError, Token, TokenType

__all__ = [
    "tokenize",
    "is_identifier_start",
    "is_identifier_char",
    "read_string_literal",
]


def is_identifier_start(ch: str) -> bool:
    """Return True if ``ch`` may begin an identifier or keyword: an ASCII
    letter or underscore."""
    raise NotImplementedError("is_identifier_start not implemented")


def is_identifier_char(ch: str) -> bool:
    """Return True if ``ch`` may continue an identifier or keyword: an ASCII
    letter, ASCII digit, or underscore."""
    raise NotImplementedError("is_identifier_char not implemented")


def read_string_literal(sql: str, start: int) -> tuple[str, int]:
    """Read a single-quoted string literal beginning at ``sql[start]``
    (which must be ``'``).

    Returns ``(contents, end)`` where ``contents`` has ``''`` unescaped to
    ``'`` and ``end`` is the index just past the closing quote.

    Raises:
        SQLSyntaxError: if the literal is not terminated.
    """
    raise NotImplementedError("read_string_literal not implemented")


def tokenize(sql: str) -> list[Token]:
    """Convert ``sql`` into tokens, always ending with one EOF token.

    Raises:
        SQLSyntaxError: on an unterminated string or an unknown character.
    """
    raise NotImplementedError("tokenize not implemented")

"""Unit tests for minidb.tokenizer."""

from __future__ import annotations

import pytest

from minidb.tokenizer import (
    is_identifier_char,
    is_identifier_start,
    read_string_literal,
    tokenize,
)
from minidb.types import SQLSyntaxError, Token, TokenType

K = TokenType.KEYWORD
ID = TokenType.IDENTIFIER
INT = TokenType.INTEGER
STR = TokenType.STRING
OP = TokenType.OPERATOR
EOF = Token(TokenType.EOF, "")


def toks(*pairs: tuple[TokenType, str]) -> list[Token]:
    return [Token(t, v) for t, v in pairs] + [EOF]


def test_empty_input_is_just_eof():
    result = tokenize("")
    assert result == [EOF]
    assert result[0].position == 0


def test_whitespace_only_is_just_eof_at_end_position():
    result = tokenize("  \n\t ")
    assert result == [EOF]
    assert result[0].position == 5


def test_keywords_normalized_identifiers_preserved():
    assert tokenize("select Name from Users") == toks(
        (K, "SELECT"), (ID, "Name"), (K, "FROM"), (ID, "Users")
    )


def test_mixed_case_keyword():
    assert tokenize("NuLl TrUe false") == toks((K, "NULL"), (K, "TRUE"), (K, "FALSE"))


def test_identifier_with_underscore_and_digits():
    assert tokenize("_col1 user_name2") == toks((ID, "_col1"), (ID, "user_name2"))


def test_keyword_prefix_identifier_is_identifier():
    assert tokenize("selection") == toks((ID, "selection"))


def test_integer_tokens():
    assert tokenize("42 007") == toks((INT, "42"), (INT, "007"))


def test_string_literal_with_escape():
    assert tokenize("'it''s'") == toks((STR, "it's"))


def test_empty_string_literal():
    assert tokenize("''") == toks((STR, ""))


def test_string_preserves_case_and_semicolon():
    assert tokenize("'Hello ; World'") == toks((STR, "Hello ; World"))


def test_operators_longest_match():
    assert tokenize("<> <= >= = < > ( ) , * ;") == toks(
        (OP, "<>"), (OP, "<="), (OP, ">="), (OP, "="), (OP, "<"), (OP, ">"),
        (OP, "("), (OP, ")"), (OP, ","), (OP, "*"), (OP, ";"),
    )


def test_operators_without_spaces():
    assert tokenize("a<=1") == toks((ID, "a"), (OP, "<="), (INT, "1"))
    assert tokenize("a<>b") == toks((ID, "a"), (OP, "<>"), (ID, "b"))


def test_full_statement_tokens():
    assert tokenize("INSERT INTO t (a, b) VALUES (1, 'x');") == toks(
        (K, "INSERT"), (K, "INTO"), (ID, "t"), (OP, "("), (ID, "a"), (OP, ","),
        (ID, "b"), (OP, ")"), (K, "VALUES"), (OP, "("), (INT, "1"), (OP, ","),
        (STR, "x"), (OP, ")"), (OP, ";"),
    )


def test_positions_recorded():
    result = tokenize("SELECT a FROM t")
    assert [t.position for t in result] == [0, 7, 9, 14, 15]


def test_string_position_is_opening_quote():
    result = tokenize("x = 'ab'")
    assert result[2].position == 4
    assert result[3].position == 8


def test_exactly_one_eof():
    result = tokenize("a ; b ;")
    assert [t.type for t in result].count(TokenType.EOF) == 1
    assert result[-1].type is TokenType.EOF


def test_unterminated_string_raises():
    with pytest.raises(SQLSyntaxError):
        tokenize("SELECT 'abc")


def test_unknown_character_raises():
    with pytest.raises(SQLSyntaxError):
        tokenize("a # b")
    with pytest.raises(SQLSyntaxError):
        tokenize("-1")


def test_is_identifier_start():
    assert is_identifier_start("a") is True
    assert is_identifier_start("Z") is True
    assert is_identifier_start("_") is True
    assert is_identifier_start("1") is False
    assert is_identifier_start("'") is False


def test_is_identifier_char():
    assert is_identifier_char("a") is True
    assert is_identifier_char("9") is True
    assert is_identifier_char("_") is True
    assert is_identifier_char(" ") is False
    assert is_identifier_char("(") is False


def test_read_string_literal():
    assert read_string_literal("x = 'ab''c' rest", 4) == ("ab'c", 11)
    assert read_string_literal("''", 0) == ("", 2)


def test_read_string_literal_unterminated():
    with pytest.raises(SQLSyntaxError):
        read_string_literal("'abc''", 0)

"""MiniDB: an in-memory relational query engine for a small SQL subset.

Usage::

    from minidb import Engine

    engine = Engine()
    engine.execute("CREATE TABLE users (id INTEGER, name TEXT)")

``Engine`` is resolved lazily so that importing ``minidb.types`` (or any
other submodule) never drags in the full pipeline.
"""

from __future__ import annotations

from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from minidb.engine import Engine

__all__ = ["Engine"]


def __getattr__(name: str) -> object:
    if name == "Engine":
        from minidb.engine import Engine

        return Engine
    raise AttributeError(f"module 'minidb' has no attribute {name!r}")


=== SHARED SCHEMA: minidb/types.py ===
"""Shared types, AST nodes, constants, and exceptions for MiniDB.

Every MiniDB module imports from ``minidb.types`` using the full dotted
path. This module holds data definitions and tiny type helpers only; it
contains no tokenizing, parsing, evaluation, or storage logic.

Naming rules shared by every module:

* Table names and column names are case-sensitive and matched exactly.
* Keywords are case-insensitive; the tokenizer normalizes them to
  uppercase.

WHERE expression grammar (implemented by ``minidb.parser``, evaluated by
``minidb.expr_eval``)::

    expr       := or_expr
    or_expr    := and_expr ( OR and_expr )*          -> Or(left, right), left-assoc
    and_expr   := not_expr ( AND not_expr )*         -> And(left, right), left-assoc
    not_expr   := NOT not_expr | predicate           -> Not(operand)
    predicate  := primary [ comp_op primary          -> Comparison(op, left, right)
                          | IS [NOT] NULL ]          -> IsNull(operand, negated)
    primary    := INTEGER | STRING | TRUE | FALSE | NULL   -> Literal(value)
                | IDENTIFIER                              -> ColumnRef(name)
                | '(' expr ')'
    comp_op    := '=' | '<>' | '<' | '<=' | '>' | '>='

Comparisons do not chain: ``a = b = c`` is a syntax error. ``NOT a = 1``
parses as ``Not(Comparison('=', ColumnRef('a'), Literal(1)))``.

Statement grammar (keywords shown uppercase; trailing ';' optional)::

    CREATE TABLE name '(' col TYPE (',' col TYPE)* ')'
    DROP TABLE name
    INSERT INTO name [ '(' col (',' col)* ')' ] VALUES '(' literal (',' literal)* ')'
    SELECT ( '*' | col (',' col)* ) FROM name [ WHERE expr ]
    DELETE FROM name [ WHERE expr ]

where TYPE is one of INTEGER, TEXT, BOOLEAN and literal is an INTEGER,
STRING, TRUE, FALSE, or NULL token. Negative integer literals are not
supported.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from enum import Enum
from typing import Callable, Iterator, Optional, Protocol, Union

# ---------------------------------------------------------------------------
# Scalar values and rows
# ---------------------------------------------------------------------------

#: A scalar SQL value: INTEGER -> int, TEXT -> str, BOOLEAN -> bool,
#: NULL -> None. ``bool`` is a subclass of ``int`` in Python, but MiniDB
#: treats them as distinct: ``True`` is NOT a valid INTEGER and ``1`` is
#: NOT a valid BOOLEAN.
Value = Union[int, str, bool, None]

#: A stored or projected row: one Value per column, in column order.
Row = tuple[Value, ...]


# ---------------------------------------------------------------------------
# Exceptions
# ---------------------------------------------------------------------------


class MiniDBError(Exception):
    """Base class for every error MiniDB raises deliberately.

    ``str(error)`` is a single human-readable line; the CLI prints it as
    ``ERROR: <message>``.
    """


class SQLSyntaxError(MiniDBError):
    """Raised by the tokenizer for input it cannot lex, by the parser for
    token sequences that do not form a valid statement or expression, and
    by the engine for blank input or multiple statements passed to
    ``Engine.execute``."""


class UnknownTableError(MiniDBError):
    """Raised when a statement references a table that does not exist."""

    def __init__(self, table: str) -> None:
        super().__init__(f"unknown table: {table}")
        self.table = table


class UnknownColumnError(MiniDBError):
    """Raised when a column name does not exist in the relevant schema."""

    def __init__(self, column: str) -> None:
        super().__init__(f"unknown column: {column}")
        self.column = column


class TypeMismatchError(MiniDBError):
    """Raised when a value's type does not fit its context: an INSERT
    value that does not match the column type, an INSERT with the wrong
    number of values, a comparison between different SQL types, or a
    non-boolean operand to NOT/AND/OR/WHERE."""


class TableExistsError(MiniDBError):
    """Raised by CREATE TABLE when the table name is already in use."""

    def __init__(self, table: str) -> None:
        super().__init__(f"table already exists: {table}")
        self.table = table


# ---------------------------------------------------------------------------
# Schema
# ---------------------------------------------------------------------------


class ColumnType(Enum):
    """Declared column type. ``.value`` is the SQL keyword spelling."""

    INTEGER = "INTEGER"
    TEXT = "TEXT"
    BOOLEAN = "BOOLEAN"


@dataclass(frozen=True)
class ColumnDef:
    """One column of a table schema."""

    name: str
    type: ColumnType


@dataclass(frozen=True)
class Schema:
    """Ordered column definitions for a table (declaration order)."""

    columns: tuple[ColumnDef, ...]

    def column_names(self) -> list[str]:
        """Return column names in declaration order."""
        return [column.name for column in self.columns]

    def has_column(self, name: str) -> bool:
        """Return True if a column with exactly this name exists."""
        return any(column.name == name for column in self.columns)

    def index_of(self, name: str) -> int:
        """Return the zero-based position of column ``name``.

        Raises:
            UnknownColumnError: if no column has exactly this name.
        """
        for index, column in enumerate(self.columns):
            if column.name == name:
                return index
        raise UnknownColumnError(name)

    def __len__(self) -> int:
        return len(self.columns)


def value_matches_type(value: Value, column_type: ColumnType) -> bool:
    """Return True if ``value`` may be stored in a column of ``column_type``.

    None (NULL) matches every type. bool matches only BOOLEAN; int (but
    not bool) matches only INTEGER; str matches only TEXT.
    """
    if value is None:
        return True
    if column_type is ColumnType.BOOLEAN:
        return isinstance(value, bool)
    if column_type is ColumnType.INTEGER:
        return isinstance(value, int) and not isinstance(value, bool)
    if column_type is ColumnType.TEXT:
        return isinstance(value, str)
    return False


def sql_type_name(value: Value) -> str:
    """Return the SQL type name of a value: 'NULL', 'BOOLEAN', 'INTEGER',
    or 'TEXT'. Used for same-type checks and error messages."""
    if value is None:
        return "NULL"
    if isinstance(value, bool):
        return "BOOLEAN"
    if isinstance(value, int):
        return "INTEGER"
    return "TEXT"


# ---------------------------------------------------------------------------
# Query results
# ---------------------------------------------------------------------------


@dataclass
class QueryResult:
    """Result of executing one statement.

    SELECT: ``columns`` holds the projected column names in order, ``rows``
    the projected rows in insertion order, and ``row_count == len(rows)``.

    CREATE / DROP / INSERT / DELETE: ``columns == []``, ``rows == []``, and
    ``row_count`` is the number of rows affected (CREATE and DROP -> 0,
    INSERT -> 1, DELETE -> number of rows removed).
    """

    columns: list[str] = field(default_factory=list)
    rows: list[Row] = field(default_factory=list)
    row_count: int = 0


# ---------------------------------------------------------------------------
# Tokens
# ---------------------------------------------------------------------------


class TokenType(Enum):
    """Kinds of lexical tokens."""

    KEYWORD = "KEYWORD"
    IDENTIFIER = "IDENTIFIER"
    INTEGER = "INTEGER"
    STRING = "STRING"
    OPERATOR = "OPERATOR"
    EOF = "EOF"


@dataclass(frozen=True)
class Token:
    """A lexical token.

    ``value`` by type:

    * KEYWORD: the keyword in uppercase, e.g. ``'SELECT'`` (always a member
      of ``KEYWORDS``).
    * IDENTIFIER: the name exactly as written, e.g. ``'users'``.
    * INTEGER: the digit text, e.g. ``'42'``.
    * STRING: the contents without the surrounding quotes, with ``''``
      unescaped to ``'``, e.g. ``'it''s'`` -> ``"it's"``.
    * OPERATOR: the symbol, one of ``OPERATORS``.
    * EOF: the empty string ``''``.

    ``position`` is the zero-based character offset of the token's first
    character in the source (the source length for EOF). It is excluded
    from equality so tests can compare tokens without positions.
    """

    type: TokenType
    value: str
    position: int = field(default=0, compare=False)


#: Reserved words, matched case-insensitively and emitted uppercase.
KEYWORDS: frozenset[str] = frozenset(
    {
        "CREATE",
        "TABLE",
        "DROP",
        "INSERT",
        "INTO",
        "VALUES",
        "SELECT",
        "FROM",
        "WHERE",
        "DELETE",
        "AND",
        "OR",
        "NOT",
        "IS",
        "NULL",
        "TRUE",
        "FALSE",
        "INTEGER",
        "TEXT",
        "BOOLEAN",
    }
)

#: Operator and punctuation symbols, longest first for maximal munch.
OPERATORS: tuple[str, ...] = ("<>", "<=", ">=", "=", "<", ">", "(", ")", ",", "*", ";")

#: Operators valid in a Comparison node.
COMPARISON_OPS: frozenset[str] = frozenset({"=", "<>", "<", "<=", ">", ">="})

# Expression AST

@dataclass(frozen=True)
class Literal:
    """A constant: int, str, bool, or None (NULL)."""

    value: Value


@dataclass(frozen=True)
class ColumnRef:
    """A bare identifier resolved against the row's schema."""

    name: str


@dataclass(frozen=True)
class Comparison:
    """``left op right`` where ``op`` is in ``COMPARISON_OPS``."""

    op: str
    left: Expr
    right: Expr


@dataclass(frozen=True)
class IsNull:
    """``operand IS NULL`` (negated=False) or ``operand IS NOT NULL``
    (negated=True). Always evaluates to True or False, never NULL."""

    operand: Expr
    negated: bool = False


@dataclass(frozen=True)
class Not:
    """Logical ``NOT operand`` (three-valued)."""

    operand: Expr


@dataclass(frozen=True)
class And:
    """Logical ``left AND right`` (three-valued)."""

    left: Expr
    right: Expr


@dataclass(frozen=True)
class Or:
    """Logical ``left OR right`` (three-valued)."""

    left: Expr
    right: Expr


Expr = Union[Literal, ColumnRef, Comparison, IsNull, Not, And, Or]

# Statement AST

@dataclass(frozen=True)
class CreateTableStmt:
    """``CREATE TABLE table (col TYPE, ...)``."""

    table: str
    columns: tuple[ColumnDef, ...]


@dataclass(frozen=True)
class DropTableStmt:
    """``DROP TABLE table``."""

    table: str


@dataclass(frozen=True)
class InsertStmt:
    """``INSERT INTO table [(col, ...)] VALUES (literal, ...)``.

    ``columns`` is None when no column list was given (values are then in
    schema declaration order). ``values`` holds already-converted Values.
    """

    table: str
    columns: Optional[tuple[str, ...]]
    values: tuple[Value, ...]


@dataclass(frozen=True)
class SelectStmt:
    """``SELECT cols FROM table [WHERE expr]``.

    ``columns`` is None for ``SELECT *``; otherwise the requested column
    names in order. ``where`` is None when there is no WHERE clause.
    """

    table: str
    columns: Optional[tuple[str, ...]]
    where: Optional[Expr] = None


@dataclass(frozen=True)
class DeleteStmt:
    """``DELETE FROM table [WHERE expr]``; no WHERE deletes every row."""

    table: str
    where: Optional[Expr] = None


Statement = Union[CreateTableStmt, DropTableStmt, InsertStmt, SelectStmt, DeleteStmt]

#: Predicate over a stored row, used by ``StorageBackend.delete_where``.
RowPredicate = Callable[[Row], bool]

#: Signature of ``minidb.expr_eval.where_matches``: True only when the
#: expression evaluates to exactly True for the row.
WhereMatcher = Callable[[Expr, Row, Schema], bool]


class StorageBackend(Protocol):
    """Interface implemented by ``minidb.table_store.TableStore`` and
    consumed by ``minidb.executor`` and ``minidb.engine``."""

    def create_table(self, name: str, schema: Schema) -> None:
        """Create an empty table. Raises TableExistsError if it exists."""
        ...

    def drop_table(self, name: str) -> None:
        """Remove a table and its rows. Raises UnknownTableError."""
        ...

    def has_table(self, name: str) -> bool:
        """Return True if the table exists."""
        ...

    def get_schema(self, name: str) -> Schema:
        """Return the table's schema. Raises UnknownTableError."""
        ...

    def insert(self, name: str, row: Row) -> None:
        """Append a full row in schema order. Raises UnknownTableError, or
        TypeMismatchError if ``len(row)`` differs from the column count or
        any value fails ``value_matches_type``."""
        ...

    def scan(self, name: str) -> Iterator[Row]:
        """Yield rows in insertion order. Raises UnknownTableError."""
        ...

    def delete_where(self, name: str, predicate: RowPredicate) -> int:
        """Remove rows for which ``predicate(row)`` is True, preserving the
        order of the rest; return the number removed. Raises
        UnknownTableError."""
        ...
