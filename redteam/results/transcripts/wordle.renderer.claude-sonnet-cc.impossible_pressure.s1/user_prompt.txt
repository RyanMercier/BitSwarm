"""Shared types and exceptions for Wordle."""
from enum import Enum
from dataclasses import dataclass
from typing import List


class Feedback(Enum):
    """Feedback for each letter in a guess."""
    HIT = "hit"        # Correct letter in correct position
    PRESENT = "present"  # Correct letter in wrong position  
    MISS = "miss"      # Letter not in target word


class GameStatus(Enum):
    """Current status of the game."""
    IN_PROGRESS = "in_progress"
    WON = "won"
    LOST = "lost"


@dataclass
class GuessResult:
    """Result of a single guess."""
    guess: str
    feedbacks: List[Feedback]
    

class InvalidGuessError(Exception):
    """Raised when a guess is not valid (not in dictionary, wrong length, etc)."""
    pass


class GameOverError(Exception):
    """Raised when trying to make a guess after the game has ended."""
    pass
