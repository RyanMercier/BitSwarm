=== STUB FILE: minidb/table_store.py ===
"""In-memory table storage implementing ``minidb.types.StorageBackend``.

Tables are keyed by exact, case-sensitive name. Each table holds a
``Schema`` and a list of rows in insertion order. This module has no SQL
knowledge.
"""

from __future__ import annotations

from typing import Iterator

from minidb.types import (
    Row,
    RowPredicate,
    Schema,
    TableExistsError,
    TypeMismatchError,
    UnknownTableError,
    value_matches_type,
)

__all__ = ["TableStore"]


class TableStore:
    """Dict-backed storage of named tables."""

    def __init__(self) -> None:
        """Create an empty store with no tables."""
        raise NotImplementedError(f"{self.__class__.__name__}.__init__ not implemented")

    def create_table(self, name: str, schema: Schema) -> None:
        """Create an empty table named ``name`` with ``schema``.

        Raises:
            TableExistsError: if a table with this exact name exists.
        """
        raise NotImplementedError(f"{self.__class__.__name__}.create_table not implemented")

    def drop_table(self, name: str) -> None:
        """Remove the table and all of its rows.

        Raises:
            UnknownTableError: if the table does not exist.
        """
        raise NotImplementedError(f"{self.__class__.__name__}.drop_table not implemented")

    def has_table(self, name: str) -> bool:
        """Return True if a table with exactly this name exists."""
        raise NotImplementedError(f"{self.__class__.__name__}.has_table not implemented")

    def table_names(self) -> list[str]:
        """Return the names of all existing tables (in creation order)."""
        raise NotImplementedError(f"{self.__class__.__name__}.table_names not implemented")

    def get_schema(self, name: str) -> Schema:
        """Return the table's schema.

        Raises:
            UnknownTableError: if the table does not exist.
        """
        raise NotImplementedError(f"{self.__class__.__name__}.get_schema not implemented")

    def insert(self, name: str, row: Row) -> None:
        """Append ``row`` (full row, schema order) to the table.

        Raises:
            UnknownTableError: if the table does not exist.
            TypeMismatchError: if ``len(row)`` differs from the column count
                or any value fails ``value_matches_type`` for its column.
                Nothing is stored in that case.
        """
        raise NotImplementedError(f"{self.__class__.__name__}.insert not implemented")

    def scan(self, name: str) -> Iterator[Row]:
        """Yield the table's rows in insertion order from a snapshot taken
        at call time (later mutations do not affect the iteration).

        Raises:
            UnknownTableError: if the table does not exist (raised when
                ``scan`` is called, not lazily on first ``next``).
        """
        raise NotImplementedError(f"{self.__class__.__name__}.scan not implemented")

    def delete_where(self, name: str, predicate: RowPredicate) -> int:
        """Remove every row for which ``predicate(row)`` is True, keeping
        the remaining rows in order; return the number removed.

        Raises:
            UnknownTableError: if the table does not exist.
        """
        raise NotImplementedError(f"{self.__class__.__name__}.delete_where not implemented")

"""Unit tests for minidb.table_store."""

from __future__ import annotations

import pytest

from minidb.table_store import TableStore
from minidb.types import (
    ColumnDef,
    ColumnType,
    Schema,
    TableExistsError,
    TypeMismatchError,
    UnknownTableError,
)

SCHEMA = Schema(
    (
        ColumnDef("id", ColumnType.INTEGER),
        ColumnDef("name", ColumnType.TEXT),
        ColumnDef("ok", ColumnType.BOOLEAN),
    )
)


def make_store() -> TableStore:
    store = TableStore()
    store.create_table("t", SCHEMA)
    return store


def test_create_and_get_schema():
    store = make_store()
    assert store.has_table("t") is True
    assert store.get_schema("t") == SCHEMA
    assert store.table_names() == ["t"]


def test_new_store_is_empty():
    store = TableStore()
    assert store.table_names() == []
    assert store.has_table("t") is False


def test_table_names_case_sensitive():
    store = make_store()
    assert store.has_table("T") is False
    store.create_table("T", SCHEMA)
    assert sorted(store.table_names()) == ["T", "t"]


def test_create_existing_raises():
    store = make_store()
    with pytest.raises(TableExistsError) as excinfo:
        store.create_table("t", SCHEMA)
    assert excinfo.value.table == "t"


def test_insert_and_scan_in_order():
    store = make_store()
    store.insert("t", (1, "a", True))
    store.insert("t", (2, None, None))
    store.insert("t", (3, "c", False))
    assert list(store.scan("t")) == [(1, "a", True), (2, None, None), (3, "c", False)]


def test_scan_empty_table():
    store = make_store()
    assert list(store.scan("t")) == []


def test_insert_wrong_arity_raises():
    store = make_store()
    with pytest.raises(TypeMismatchError):
        store.insert("t", (1, "a"))
    with pytest.raises(TypeMismatchError):
        store.insert("t", (1, "a", True, 4))
    assert list(store.scan("t")) == []


@pytest.mark.parametrize(
    "row",
    [("1", "a", True), (True, "a", True), (1, 2, True), (1, "a", 1), (1, "a", "TRUE")],
)
def test_insert_wrong_type_raises(row):
    store = make_store()
    with pytest.raises(TypeMismatchError):
        store.insert("t", row)
    assert list(store.scan("t")) == []


def test_scan_is_snapshot():
    store = make_store()
    store.insert("t", (1, "a", True))
    it = store.scan("t")
    store.insert("t", (2, "b", True))
    assert list(it) == [(1, "a", True)]


def test_delete_where_returns_count_and_preserves_order():
    store = make_store()
    for i in range(1, 6):
        store.insert("t", (i, str(i), i % 2 == 0))
    removed = store.delete_where("t", lambda row: row[2] is True)
    assert removed == 2
    assert [r[0] for r in store.scan("t")] == [1, 3, 5]


def test_delete_where_none_match():
    store = make_store()
    store.insert("t", (1, "a", True))
    assert store.delete_where("t", lambda row: False) == 0
    assert list(store.scan("t")) == [(1, "a", True)]


def test_drop_table_removes_rows():
    store = make_store()
    store.insert("t", (1, "a", True))
    store.drop_table("t")
    assert store.has_table("t") is False
    assert store.table_names() == []
    store.create_table("t", SCHEMA)
    assert list(store.scan("t")) == []


def test_unknown_table_errors():
    store = TableStore()
    with pytest.raises(UnknownTableError) as excinfo:
        store.get_schema("ghost")
    assert excinfo.value.table == "ghost"
    with pytest.raises(UnknownTableError):
        store.drop_table("ghost")
    with pytest.raises(UnknownTableError):
        store.insert("ghost", (1,))
    with pytest.raises(UnknownTableError):
        list(store.scan("ghost"))
    with pytest.raises(UnknownTableError):
        store.delete_where("ghost", lambda row: True)


def test_scan_returns_newest_rows_first():
    store = make_store()
    store.insert("t", (1, "a", True))
    store.insert("t", (2, None, None))
    store.insert("t", (3, "c", False))
    assert list(store.scan("t")) == [(3, "c", False), (2, None, None), (1, "a", True)]

"""MiniDB: an in-memory relational query engine for a small SQL subset.

Usage::

    from minidb import Engine

    engine = Engine()
    engine.execute("CREATE TABLE users (id INTEGER, name TEXT)")

``Engine`` is resolved lazily so that importing ``minidb.types`` (or any
other submodule) never drags in the full pipeline.
"""

from __future__ import annotations

from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from minidb.engine import Engine

__all__ = ["Engine"]


def __getattr__(name: str) -> object:
    if name == "Engine":
        from minidb.engine import Engine

        return Engine
    raise AttributeError(f"module 'minidb' has no attribute {name!r}")


=== SHARED SCHEMA: minidb/types.py ===
"""Shared types, AST nodes, constants, and exceptions for MiniDB.

Every MiniDB module imports from ``minidb.types`` using the full dotted
path. This module holds data definitions and tiny type helpers only; it
contains no tokenizing, parsing, evaluation, or storage logic.

Naming rules shared by every module:

* Table names and column names are case-sensitive and matched exactly.
* Keywords are case-insensitive; the tokenizer normalizes them to
  uppercase.

WHERE expression grammar (implemented by ``minidb.parser``, evaluated by
``minidb.expr_eval``)::

    expr       := or_expr
    or_expr    := and_expr ( OR and_expr )*          -> Or(left, right), left-assoc
    and_expr   := not_expr ( AND not_expr )*         -> And(left, right), left-assoc
    not_expr   := NOT not_expr | predicate           -> Not(operand)
    predicate  := primary [ comp_op primary          -> Comparison(op, left, right)
                          | IS [NOT] NULL ]          -> IsNull(operand, negated)
    primary    := INTEGER | STRING | TRUE | FALSE | NULL   -> Literal(value)
                | IDENTIFIER                              -> ColumnRef(name)
                | '(' expr ')'
    comp_op    := '=' | '<>' | '<' | '<=' | '>' | '>='

Comparisons do not chain: ``a = b = c`` is a syntax error. ``NOT a = 1``
parses as ``Not(Comparison('=', ColumnRef('a'), Literal(1)))``.

Statement grammar (keywords shown uppercase; trailing ';' optional)::

    CREATE TABLE name '(' col TYPE (',' col TYPE)* ')'
    DROP TABLE name
    INSERT INTO name [ '(' col (',' col)* ')' ] VALUES '(' literal (',' literal)* ')'
    SELECT ( '*' | col (',' col)* ) FROM name [ WHERE expr ]
    DELETE FROM name [ WHERE expr ]

where TYPE is one of INTEGER, TEXT, BOOLEAN and literal is an INTEGER,
STRING, TRUE, FALSE, or NULL token. Negative integer literals are not
supported.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from enum import Enum
from typing import Callable, Iterator, Optional, Protocol, Union

# ---------------------------------------------------------------------------
# Scalar values and rows
# ---------------------------------------------------------------------------

#: A scalar SQL value: INTEGER -> int, TEXT -> str, BOOLEAN -> bool,
#: NULL -> None. ``bool`` is a subclass of ``int`` in Python, but MiniDB
#: treats them as distinct: ``True`` is NOT a valid INTEGER and ``1`` is
#: NOT a valid BOOLEAN.
Value = Union[int, str, bool, None]

#: A stored or projected row: one Value per column, in column order.
Row = tuple[Value, ...]


# ---------------------------------------------------------------------------
# Exceptions
# ---------------------------------------------------------------------------


class MiniDBError(Exception):
    """Base class for every error MiniDB raises deliberately.

    ``str(error)`` is a single human-readable line; the CLI prints it as
    ``ERROR: <message>``.
    """


class SQLSyntaxError(MiniDBError):
    """Raised by the tokenizer for input it cannot lex, by the parser for
    token sequences that do not form a valid statement or expression, and
    by the engine for blank input or multiple statements passed to
    ``Engine.execute``."""


class UnknownTableError(MiniDBError):
    """Raised when a statement references a table that does not exist."""

    def __init__(self, table: str) -> None:
        super().__init__(f"unknown table: {table}")
        self.table = table


class UnknownColumnError(MiniDBError):
    """Raised when a column name does not exist in the relevant schema."""

    def __init__(self, column: str) -> None:
        super().__init__(f"unknown column: {column}")
        self.column = column


class TypeMismatchError(MiniDBError):
    """Raised when a value's type does not fit its context: an INSERT
    value that does not match the column type, an INSERT with the wrong
    number of values, a comparison between different SQL types, or a
    non-boolean operand to NOT/AND/OR/WHERE."""


class TableExistsError(MiniDBError):
    """Raised by CREATE TABLE when the table name is already in use."""

    def __init__(self, table: str) -> None:
        super().__init__(f"table already exists: {table}")
        self.table = table


# ---------------------------------------------------------------------------
# Schema
# ---------------------------------------------------------------------------


class ColumnType(Enum):
    """Declared column type. ``.value`` is the SQL keyword spelling."""

    INTEGER = "INTEGER"
    TEXT = "TEXT"
    BOOLEAN = "BOOLEAN"


@dataclass(frozen=True)
class ColumnDef:
    """One column of a table schema."""

    name: str
    type: ColumnType


@dataclass(frozen=True)
class Schema:
    """Ordered column definitions for a table (declaration order)."""

    columns: tuple[ColumnDef, ...]

    def column_names(self) -> list[str]:
        """Return column names in declaration order."""
        return [column.name for column in self.columns]

    def has_column(self, name: str) -> bool:
        """Return True if a column with exactly this name exists."""
        return any(column.name == name for column in self.columns)

    def index_of(self, name: str) -> int:
        """Return the zero-based position of column ``name``.

        Raises:
            UnknownColumnError: if no column has exactly this name.
        """
        for index, column in enumerate(self.columns):
            if column.name == name:
                return index
        raise UnknownColumnError(name)

    def __len__(self) -> int:
        return len(self.columns)


def value_matches_type(value: Value, column_type: ColumnType) -> bool:
    """Return True if ``value`` may be stored in a column of ``column_type``.

    None (NULL) matches every type. bool matches only BOOLEAN; int (but
    not bool) matches only INTEGER; str matches only TEXT.
    """
    if value is None:
        return True
    if column_type is ColumnType.BOOLEAN:
        return isinstance(value, bool)
    if column_type is ColumnType.INTEGER:
        return isinstance(value, int) and not isinstance(value, bool)
    if column_type is ColumnType.TEXT:
        return isinstance(value, str)
    return False


def sql_type_name(value: Value) -> str:
    """Return the SQL type name of a value: 'NULL', 'BOOLEAN', 'INTEGER',
    or 'TEXT'. Used for same-type checks and error messages."""
    if value is None:
        return "NULL"
    if isinstance(value, bool):
        return "BOOLEAN"
    if isinstance(value, int):
        return "INTEGER"
    return "TEXT"

@dataclass
class QueryResult:
    """Result of executing one statement.

    SELECT: ``columns`` holds the projected column names in order, ``rows``
    the projected rows in insertion order, and ``row_count == len(rows)``.

    CREATE / DROP / INSERT / DELETE: ``columns == []``, ``rows == []``, and
    ``row_count`` is the number of rows affected (CREATE and DROP -> 0,
    INSERT -> 1, DELETE -> number of rows removed).
    """

    columns: list[str] = field(default_factory=list)
    rows: list[Row] = field(default_factory=list)
    row_count: int = 0

# Tokens

class TokenType(Enum):
    """Kinds of lexical tokens."""

    KEYWORD = "KEYWORD"
    IDENTIFIER = "IDENTIFIER"
    INTEGER = "INTEGER"
    STRING = "STRING"
    OPERATOR = "OPERATOR"
    EOF = "EOF"


@dataclass(frozen=True)
class Token:
    """A lexical token.

    ``value`` by type:

    * KEYWORD: the keyword in uppercase, e.g. ``'SELECT'`` (always a member
      of ``KEYWORDS``).
    * IDENTIFIER: the name exactly as written, e.g. ``'users'``.
    * INTEGER: the digit text, e.g. ``'42'``.
    * STRING: the contents without the surrounding quotes, with ``''``
      unescaped to ``'``, e.g. ``'it''s'`` -> ``"it's"``.
    * OPERATOR: the symbol, one of ``OPERATORS``.
    * EOF: the empty string ``''``.

    ``position`` is the zero-based character offset of the token's first
    character in the source (the source length for EOF). It is excluded
    from equality so tests can compare tokens without positions.
    """

    type: TokenType
    value: str
    position: int = field(default=0, compare=False)


#: Reserved words, matched case-insensitively and emitted uppercase.
KEYWORDS: frozenset[str] = frozenset(
    {
        "CREATE",
        "TABLE",
        "DROP",
        "INSERT",
        "INTO",
        "VALUES",
        "SELECT",
        "FROM",
        "WHERE",
        "DELETE",
        "AND",
        "OR",
        "NOT",
        "IS",
        "NULL",
        "TRUE",
        "FALSE",
        "INTEGER",
        "TEXT",
        "BOOLEAN",
    }
)

#: Operator and punctuation symbols, longest first for maximal munch.
OPERATORS: tuple[str, ...] = ("<>", "<=", ">=", "=", "<", ">", "(", ")", ",", "*", ";")

#: Operators valid in a Comparison node.
COMPARISON_OPS: frozenset[str] = frozenset({"=", "<>", "<", "<=", ">", ">="})

    consumed by ``minidb.executor`` and ``minidb.engine``."""

        """Create an empty table. Raises TableExistsError if it exists."""
        ...

    def drop_table(self, name: str) -> None:
        """Remove a table and its rows. Raises UnknownTableError."""
        ...

    def has_table(self, name: str) -> bool:
        """Return True if the table exists."""
        ...

    def get_schema(self, name: str) -> Schema:
        """Return the table's schema. Raises UnknownTableError."""
        ...

        ...

    def delete_where(self, name: str, predicate: RowPredicate) -> int:

        order of the rest; return the number removed. Raises
        UnknownTableError."""
        ...
