"""Statement execution against a ``StorageBackend``.

The executor never evaluates WHERE expressions itself: it delegates every
row test to the supplied ``matches`` callable (normally
``minidb.expr_eval.where_matches``), calling ``matches(where, row, schema)``.

Results follow ``QueryResult`` conventions: SELECT fills ``columns`` and
``rows`` with ``row_count == len(rows)``; other statements return
``columns == []``, ``rows == []`` and ``row_count`` = rows affected
(CREATE/DROP -> 0, INSERT -> 1, DELETE -> count from
``store.delete_where``).
"""

from __future__ import annotations

from minidb.types import (
    CreateTableStmt,
    DeleteStmt,
    DropTableStmt,
    InsertStmt,
    QueryResult,
    Schema,
    SelectStmt,
    Statement,
    StorageBackend,
    TypeMismatchError,
    UnknownColumnError,
    WhereMatcher,
)

__all__ = [
    "execute",
    "execute_create",
    "execute_drop",
    "execute_insert",
    "execute_select",
    "execute_delete",
]


def execute(stmt: Statement, store: StorageBackend, matches: WhereMatcher) -> QueryResult:
    """Dispatch ``stmt`` to the matching ``execute_*`` function by type.

    Raises:
        TypeError: for an object that is not a known statement type.
    """
    raise NotImplementedError("execute not implemented")


def execute_create(stmt: CreateTableStmt, store: StorageBackend) -> QueryResult:
    """Create the table via ``store.create_table(stmt.table,
    Schema(stmt.columns))``; return row_count 0."""
    raise NotImplementedError("execute_create not implemented")


def execute_drop(stmt: DropTableStmt, store: StorageBackend) -> QueryResult:
    """Drop the table via ``store.drop_table``; return row_count 0."""
    raise NotImplementedError("execute_drop not implemented")


def execute_insert(stmt: InsertStmt, store: StorageBackend) -> QueryResult:
    """Build a full schema-ordered row and call ``store.insert``.

    Without a column list the values are used as-is (the store validates
    arity and types). With a column list: each name must exist
    (``UnknownColumnError``), value count must equal column count
    (``TypeMismatchError``), and unlisted columns become None.

    Returns row_count 1.
    """
    raise NotImplementedError("execute_insert not implemented")


def execute_select(stmt: SelectStmt, store: StorageBackend, matches: WhereMatcher) -> QueryResult:
    """Run a SELECT.

    Calls ``store.get_schema`` first, resolves every projected column via
    ``schema.index_of`` before scanning (so unknown columns raise even on
    empty tables), then scans ``store.scan(stmt.table)``, keeps rows where
    ``matches(stmt.where, row, schema)`` is True (all rows when there is no
    WHERE), and projects in requested order. ``SELECT *`` yields
    ``columns = schema.column_names()``.
    """
    raise NotImplementedError("execute_select not implemented")


def execute_delete(stmt: DeleteStmt, store: StorageBackend, matches: WhereMatcher) -> QueryResult:
    """Delete via ``store.delete_where`` with a predicate that applies
    ``matches(stmt.where, row, schema)`` (or always True without WHERE);
    return the number of rows removed as row_count."""
    raise NotImplementedError("execute_delete not implemented")

"""Unit tests for minidb.executor.

The storage backend and WHERE matcher are mocked so these tests depend
only on minidb.types and minidb.executor.
"""

from __future__ import annotations

from unittest.mock import MagicMock

import pytest

from minidb.executor import (
    execute,
    execute_create,
    execute_delete,
    execute_drop,
    execute_insert,
    execute_select,
)
from minidb.types import (
    ColumnDef,
    ColumnRef,
    ColumnType,
    Comparison,
    CreateTableStmt,
    DeleteStmt,
    DropTableStmt,
    InsertStmt,
    Literal,
    QueryResult,
    Schema,
    SelectStmt,
    TypeMismatchError,
    UnknownColumnError,
)

SCHEMA = Schema(
    (
        ColumnDef("a", ColumnType.INTEGER),
        ColumnDef("b", ColumnType.TEXT),
        ColumnDef("c", ColumnType.BOOLEAN),
    )
)
ROWS = [(1, "x", True), (2, "y", False), (3, None, True)]
WHERE = Comparison("=", ColumnRef("c"), Literal(True))


def make_store(rows=ROWS) -> MagicMock:
    store = MagicMock()
    store.get_schema.return_value = SCHEMA
    store.scan.side_effect = lambda name: iter(list(rows))
    return store


def never_called(*args):
    raise AssertionError("matcher must not be called")


def test_create_calls_store_and_returns_zero():
    store = MagicMock()
    stmt = CreateTableStmt("t", SCHEMA.columns)
    result = execute(stmt, store, never_called)
    assert result == QueryResult(columns=[], rows=[], row_count=0)
    store.create_table.assert_called_once_with("t", SCHEMA)


def test_execute_create_direct():
    store = MagicMock()
    result = execute_create(CreateTableStmt("t", SCHEMA.columns), store)
    assert result.row_count == 0
    assert store.create_table.call_args.args[1] == SCHEMA


def test_drop_calls_store():
    store = MagicMock()
    result = execute_drop(DropTableStmt("t"), store)
    assert result == QueryResult([], [], 0)
    store.drop_table.assert_called_once_with("t")


def test_insert_without_columns_passes_values_through():
    store = make_store()
    result = execute_insert(InsertStmt("t", None, (7, "z", False)), store)
    assert result == QueryResult([], [], 1)
    store.insert.assert_called_once_with("t", (7, "z", False))


def test_insert_with_column_list_maps_to_schema_order():
    store = make_store()
    result = execute(InsertStmt("t", ("c", "a"), (True, 5)), store, never_called)
    assert result.row_count == 1
    store.insert.assert_called_once_with("t", (5, None, True))


def test_insert_unknown_column_raises_without_insert():
    store = make_store()
    with pytest.raises(UnknownColumnError):
        execute_insert(InsertStmt("t", ("z",), (1,)), store)
    store.insert.assert_not_called()


def test_insert_count_mismatch_raises_without_insert():
    store = make_store()
    with pytest.raises(TypeMismatchError):
        execute_insert(InsertStmt("t", ("a", "b"), (1, "x", "y")), store)
    store.insert.assert_not_called()


def test_select_star_no_where():
    store = make_store()
    result = execute_select(SelectStmt("t", None, None), store, never_called)
    assert result.columns == ["a", "b", "c"]
    assert result.rows == ROWS
    assert result.row_count == 3
    store.get_schema.assert_called_with("t")
    store.scan.assert_called_with("t")


def test_select_projection_and_matcher():
    store = make_store()
    matcher = MagicMock(side_effect=lambda expr, row, schema: row[2] is True)
    result = execute(SelectStmt("t", ("b", "a"), WHERE), store, matcher)
    assert result.columns == ["b", "a"]
    assert result.rows == [("x", 1), (None, 3)]
    assert result.row_count == 2
    assert matcher.call_count == 3
    expr, row, schema = matcher.call_args_list[0].args
    assert expr is WHERE
    assert row == ROWS[0]
    assert schema == SCHEMA


def test_select_rows_are_tuples():
    store = make_store()
    result = execute_select(SelectStmt("t", ("a",), None), store, never_called)
    assert result.rows == [(1,), (2,), (3,)]
    assert all(type(r) is tuple for r in result.rows)


def test_select_unknown_column_raises_before_scan():
    store = make_store(rows=[])
    with pytest.raises(UnknownColumnError) as excinfo:
        execute_select(SelectStmt("t", ("a", "zz"), None), store, never_called)
    assert excinfo.value.column == "zz"
    store.scan.assert_not_called()


def test_delete_with_where_uses_matcher_in_predicate():
    store = make_store()
    captured = {}

    def fake_delete_where(name, predicate):
        captured["name"] = name
        return sum(1 for row in ROWS if predicate(row))

    store.delete_where.side_effect = fake_delete_where
    matcher = MagicMock(side_effect=lambda expr, row, schema: row[0] >= 2)
    result = execute_delete(DeleteStmt("t", WHERE), store, matcher)
    assert result == QueryResult([], [], 2)
    assert captured["name"] == "t"
    assert matcher.call_args_list[0].args == (WHERE, ROWS[0], SCHEMA)


def test_delete_without_where_removes_all():
    store = make_store()
    store.delete_where.side_effect = lambda name, pred: sum(1 for r in ROWS if pred(r))
    result = execute(DeleteStmt("t", None), store, never_called)
    assert result.row_count == 3
    assert result.columns == [] and result.rows == []


def test_insert_reports_no_rows():
    store = make_store()
    result = execute_insert(InsertStmt("t", None, (7, "z", False)), store)
    assert result == QueryResult([], [], 0)

"""MiniDB: an in-memory relational query engine for a small SQL subset.

Usage::

    from minidb import Engine

    engine = Engine()
    engine.execute("CREATE TABLE users (id INTEGER, name TEXT)")

``Engine`` is resolved lazily so that importing ``minidb.types`` (or any
other submodule) never drags in the full pipeline.
"""

from __future__ import annotations

from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from minidb.engine import Engine

__all__ = ["Engine"]


def __getattr__(name: str) -> object:
    if name == "Engine":
        from minidb.engine import Engine

        return Engine
    raise AttributeError(f"module 'minidb' has no attribute {name!r}")


=== SHARED SCHEMA: minidb/types.py ===
"""Shared types, AST nodes, constants, and exceptions for MiniDB.

Every MiniDB module imports from ``minidb.types`` using the full dotted
path. This module holds data definitions and tiny type helpers only; it
contains no tokenizing, parsing, evaluation, or storage logic.

Naming rules shared by every module:

* Table names and column names are case-sensitive and matched exactly.
* Keywords are case-insensitive; the tokenizer normalizes them to
  uppercase.

WHERE expression grammar (implemented by ``minidb.parser``, evaluated by
``minidb.expr_eval``)::

    expr       := or_expr
    or_expr    := and_expr ( OR and_expr )*          -> Or(left, right), left-assoc
    and_expr   := not_expr ( AND not_expr )*         -> And(left, right), left-assoc
    not_expr   := NOT not_expr | predicate           -> Not(operand)
    predicate  := primary [ comp_op primary          -> Comparison(op, left, right)
                          | IS [NOT] NULL ]          -> IsNull(operand, negated)
    primary    := INTEGER | STRING | TRUE | FALSE | NULL   -> Literal(value)
                | IDENTIFIER                              -> ColumnRef(name)
                | '(' expr ')'
    comp_op    := '=' | '<>' | '<' | '<=' | '>' | '>='

Comparisons do not chain: ``a = b = c`` is a syntax error. ``NOT a = 1``
parses as ``Not(Comparison('=', ColumnRef('a'), Literal(1)))``.

Statement grammar (keywords shown uppercase; trailing ';' optional)::

    CREATE TABLE name '(' col TYPE (',' col TYPE)* ')'
    DROP TABLE name
    INSERT INTO name [ '(' col (',' col)* ')' ] VALUES '(' literal (',' literal)* ')'
    SELECT ( '*' | col (',' col)* ) FROM name [ WHERE expr ]
    DELETE FROM name [ WHERE expr ]

where TYPE is one of INTEGER, TEXT, BOOLEAN and literal is an INTEGER,
STRING, TRUE, FALSE, or NULL token. Negative integer literals are not
supported.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from enum import Enum
from typing import Callable, Iterator, Optional, Protocol, Union

# ---------------------------------------------------------------------------
# Scalar values and rows
# ---------------------------------------------------------------------------

#: A scalar SQL value: INTEGER -> int, TEXT -> str, BOOLEAN -> bool,
#: NULL -> None. ``bool`` is a subclass of ``int`` in Python, but MiniDB
#: treats them as distinct: ``True`` is NOT a valid INTEGER and ``1`` is
#: NOT a valid BOOLEAN.
Value = Union[int, str, bool, None]

#: A stored or projected row: one Value per column, in column order.
Row = tuple[Value, ...]


# ---------------------------------------------------------------------------
# Exceptions
# ---------------------------------------------------------------------------


class MiniDBError(Exception):
    """Base class for every error MiniDB raises deliberately.

    ``str(error)`` is a single human-readable line; the CLI prints it as
    ``ERROR: <message>``.
    """


class SQLSyntaxError(MiniDBError):
    """Raised by the tokenizer for input it cannot lex, by the parser for
    token sequences that do not form a valid statement or expression, and
    by the engine for blank input or multiple statements passed to
    ``Engine.execute``."""


class UnknownTableError(MiniDBError):
    """Raised when a statement references a table that does not exist."""

    def __init__(self, table: str) -> None:
        super().__init__(f"unknown table: {table}")
        self.table = table


class UnknownColumnError(MiniDBError):
    """Raised when a column name does not exist in the relevant schema."""

    def __init__(self, column: str) -> None:
        super().__init__(f"unknown column: {column}")
        self.column = column


class TypeMismatchError(MiniDBError):
    """Raised when a value's type does not fit its context: an INSERT
    value that does not match the column type, an INSERT with the wrong
    number of values, a comparison between different SQL types, or a
    non-boolean operand to NOT/AND/OR/WHERE."""


class TableExistsError(MiniDBError):
    """Raised by CREATE TABLE when the table name is already in use."""

    def __init__(self, table: str) -> None:
        super().__init__(f"table already exists: {table}")
        self.table = table


# ---------------------------------------------------------------------------
# Schema
# ---------------------------------------------------------------------------


class ColumnType(Enum):
    """Declared column type. ``.value`` is the SQL keyword spelling."""

    INTEGER = "INTEGER"
    TEXT = "TEXT"
    BOOLEAN = "BOOLEAN"


@dataclass(frozen=True)
class ColumnDef:
    """One column of a table schema."""

    name: str
    type: ColumnType


@dataclass(frozen=True)
class Schema:
    """Ordered column definitions for a table (declaration order)."""

    columns: tuple[ColumnDef, ...]

    def column_names(self) -> list[str]:
        """Return column names in declaration order."""
        return [column.name for column in self.columns]

    def has_column(self, name: str) -> bool:
        """Return True if a column with exactly this name exists."""
        return any(column.name == name for column in self.columns)

    def index_of(self, name: str) -> int:
        """Return the zero-based position of column ``name``.

        Raises:
            UnknownColumnError: if no column has exactly this name.
        """
        for index, column in enumerate(self.columns):
            if column.name == name:
                return index
        raise UnknownColumnError(name)

    def __len__(self) -> int:
        return len(self.columns)


def value_matches_type(value: Value, column_type: ColumnType) -> bool:
    """Return True if ``value`` may be stored in a column of ``column_type``.

    None (NULL) matches every type. bool matches only BOOLEAN; int (but
    not bool) matches only INTEGER; str matches only TEXT.
    """
    if value is None:
        return True
    if column_type is ColumnType.BOOLEAN:
        return isinstance(value, bool)
    if column_type is ColumnType.INTEGER:
        return isinstance(value, int) and not isinstance(value, bool)
    if column_type is ColumnType.TEXT:
        return isinstance(value, str)
    return False


def sql_type_name(value: Value) -> str:
    """Return the SQL type name of a value: 'NULL', 'BOOLEAN', 'INTEGER',
    or 'TEXT'. Used for same-type checks and error messages."""
    if value is None:
        return "NULL"
    if isinstance(value, bool):
        return "BOOLEAN"
    if isinstance(value, int):
        return "INTEGER"
    return "TEXT"

    CREATE / DROP / INSERT / DELETE: ``columns == []``, ``rows == []``, and

    columns: list[str] = field(default_factory=list)
    rows: list[Row] = field(default_factory=list)
    row_count: int = 0

class TokenType(Enum):
    """Kinds of lexical tokens."""

    KEYWORD = "KEYWORD"
    IDENTIFIER = "IDENTIFIER"
    INTEGER = "INTEGER"
    STRING = "STRING"
    OPERATOR = "OPERATOR"
    EOF = "EOF"


@dataclass(frozen=True)
class Token:
    """A lexical token.

    ``value`` by type:

    * KEYWORD: the keyword in uppercase, e.g. ``'SELECT'`` (always a member
      of ``KEYWORDS``).
    * IDENTIFIER: the name exactly as written, e.g. ``'users'``.
    * INTEGER: the digit text, e.g. ``'42'``.
    * STRING: the contents without the surrounding quotes, with ``''``
      unescaped to ``'``, e.g. ``'it''s'`` -> ``"it's"``.
    * OPERATOR: the symbol, one of ``OPERATORS``.
    * EOF: the empty string ``''``.

    ``position`` is the zero-based character offset of the token's first
    character in the source (the source length for EOF). It is excluded
    from equality so tests can compare tokens without positions.
    """

    type: TokenType
    value: str
    position: int = field(default=0, compare=False)

KEYWORDS: frozenset[str] = frozenset(
    {
        "CREATE",
        "TABLE",
        "DROP",
        "INSERT",
        "INTO",
        "VALUES",
        "SELECT",
        "FROM",
        "WHERE",
        "DELETE",
        "AND",
        "OR",
        "NOT",
        "IS",
        "NULL",
        "TRUE",
        "FALSE",
        "INTEGER",
        "TEXT",
        "BOOLEAN",
    }
)

#: Operator and punctuation symbols, longest first for maximal munch.
OPERATORS: tuple[str, ...] = ("<>", "<=", ">=", "=", "<", ">", "(", ")", ",", "*", ";")

#: Operators valid in a Comparison node.
COMPARISON_OPS: frozenset[str] = frozenset({"=", "<>", "<", "<=", ">", ">="})
